Crear tabla principal de reconocimiento de placas
Esta tabla guardará cada detección realizada por los modelos de IA.
-- 1. Tabla Principal: Cada detección es un evento único
CREATE TABLE plate_detections (
    id INT AUTO_INCREMENT PRIMARY KEY,
    plate_text VARCHAR(10) NOT NULL,        -- Ej: 'PFJ-2048'
    yolo_confidence FLOAT,                  -- Confianza de detección de la caja
    ocr_confidence FLOAT,                   -- Confianza de lectura de EasyOCR
    
    -- Atributos de Calidad (Lo que nos da Gemini)
    is_legible ENUM('Legible', 'Ilegible') DEFAULT 'Legible',
    oclusion EN('No', 'Parcial', 'Severa') DEFAULT 'No',
    reflejo ENUM('No', 'Sí') DEFAULT 'No',
    sucia ENUM('No', 'Sí') DEFAULT 'No',
    
    vehicle_type VARCHAR(20) DEFAULT 'Desconocido', -- Auto, Moto, etc.
    image_path TEXT,                        -- Ruta al archivo en el servidor
    
    detection_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Ejemplo de registro real:

✔ Identificación → detectar y leer la placa
✔ Clasificación → clasificar vehículo / tipo de placa

Auditorias:
CREATE TABLE plate_audit_logs (
    id INT AUTO_INCREMENT PRIMARY KEY,
    plate_detection_id INT,
    checked_by VARCHAR(50),                 -- Usuario que consultó
    check_reason TEXT,                      -- Motivo (ej: 'Control de rutina')
    check_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (plate_detection_id) REFERENCES plate_detections(id)
);

Tabla para imágenes almacenadas
CREATE TABLE plate_images (
    id INT AUTO_INCREMENT PRIMARY KEY,
    
    plate_detection_id INT,
    image_path TEXT,
    
    uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (plate_detection_id) REFERENCES plate_detections(id)
);
